import 'package:flutter/material.dart';

import 'common_dropdown_item.txt';
import 'common_input_style.dart';

class CommonDropdownTile<T> extends StatelessWidget {
  final CommonDropdownItem<T> item;
  final ValueChanged<CommonDropdownItem<T>> onSelected;
  final CommonInputStyle style;
  final TextStyle? textStyle;
  final bool selected;

  const CommonDropdownTile({
    super.key,
    required this.item,
    required this.onSelected,
    required this.style,
    this.textStyle,
    this.selected = false,
  });

  @override
  Widget build(BuildContext context) {
    final bool hasChildren = item.children.isNotEmpty;

    if (hasChildren) {
      return ExpansionTile(
        leading: item.leading,
        title: Text(item.title, style: textStyle),
        trailing: item.trailing,
        children: item.children.map((child) {
          return Padding(
            padding: const EdgeInsets.only(left: 16),
            child: CommonDropdownTile<T>(
              item: child,
              onSelected: onSelected,
              style: style,
              textStyle: textStyle,
            ),
          );
        }).toList(),
      );
    }

    return ListTile(
      leading: item.leading,
      title: Text(item.title, style: textStyle),
      trailing: item.trailing,
      onTap: () {
        item.onTap?.call();
        onSelected(item);
      },
      contentPadding: style.itemPadding,
      tileColor: style.itemBackgroundColor,
      selected: selected,
      selectedTileColor: style.selectedItemColor,
      hoverColor: style.hoverItemColor,
    );
  }
}
